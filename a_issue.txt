import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:hdfc_merchant_app/features/reports_amps/bloc/download_report_state.dart';
import 'package:hdfc_merchant_app/features/reports_amps/data/reports_repository.dart';

class DownloadReportCubit extends Cubit<DownloadReportState> {
  final ReportsRepository reportRepository;
  DownloadReportCubit(this.reportRepository) : super(DownloadReportInitial());

  Future<void> downloadReport(String reportId, String scheduleName) async {
    emit(DownloadReportLoading(reportId));

    try {
      await reportRepository.downloadReport(reportId, scheduleName);

      emit(DownloadReportSuccess(reportId));

      // emit()
    } catch (e) {
      emit(DownloadReportFailure(reportId, message: 'Exception'));
    }
  }
}
